-- ==============================================================================
-- Jordan Curriculum Database Schema (Supabase PostgreSQL)
-- Updated with Specializations (التخصص), Hijri Academic Calendar, and Trial Teacher Limits
-- ==============================================================================

-- 1. جدول التقويم المدرسي والعام الهجري
CREATE TABLE IF NOT EXISTS academic_calendar (
    id SERIAL PRIMARY KEY,
    year1 INTEGER NOT NULL,                       -- السنة الميلادية الأولى (مثال: 2026)
    hijri_year TEXT NOT NULL,                     -- التاريخ الهجري للعام الدراسي (مثال: 1447 - 1448 هـ)
    sem1_start DATE NOT NULL,
    sem1_end DATE NOT NULL,
    sem2_start DATE NOT NULL,
    sem2_end DATE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. جدول التخصصات والفروع (خاص بالصفوف العاشر والحادي عشر والثاني عشر)
CREATE TABLE IF NOT EXISTS specializations (
    id BIGSERIAL PRIMARY KEY,
    grade INTEGER NOT NULL CHECK (grade IN (10, 11, 12)),
    name TEXT NOT NULL,                           -- اسم التخصص (مثال: هندسي، زراعي، علمي، أدبي، مهارات رقمية BTEC)
    code TEXT NOT NULL,                           -- رمز التخصص (مثال: ENG, AGR, SCI, LIT)
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. جدول الكتب والمناهج المدرسية (معدل لدعم التخصصات والمناهج المشتركة)
CREATE TABLE IF NOT EXISTS books (
    id BIGSERIAL PRIMARY KEY,
    grade INTEGER NOT NULL,                       -- 1 إلى 12
    semester INTEGER NOT NULL CHECK (semester IN (1, 2)),
    subject TEXT NOT NULL,                        -- اسم المبحث
    slug TEXT NOT NULL UNIQUE,
    pdf_url TEXT,
    page_offset INTEGER NOT NULL DEFAULT 0,
    units JSONB NOT NULL DEFAULT '[]'::jsonb,
    is_shared BOOLEAN DEFAULT TRUE,               -- هل المنهاج مشترك لجميع التخصصات (مثل المهارات الرقمية)
    specialization_id BIGINT REFERENCES specializations(id) ON DELETE SET NULL, -- تخصص محدد إن وجد
    specialization_code TEXT,                     -- كود التخصص (مثل 'ALL' للمشترك، أو 'ENG' للهندسي، إلخ)
    language TEXT DEFAULT 'ar' CHECK (language IN ('ar', 'en')), -- لغة المبحث (ar / en)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. جدول المستخدمين (معدل لدعم التخصص والشعبة والحسابات التجريبية)
CREATE TABLE IF NOT EXISTS users (
    id BIGSERIAL PRIMARY KEY,
    name TEXT NOT NULL,
    phone TEXT NOT NULL UNIQUE,
    email TEXT,
    password_hash TEXT NOT NULL,
    role TEXT NOT NULL CHECK (role IN ('student', 'teacher', 'supervisor', 'admin')),
    status TEXT NOT NULL DEFAULT 'active',
    grade INTEGER,                                -- الصف الدراسي
    specialization TEXT,                          -- التخصص (للصفوف 10، 11، 12)
    specialization_id BIGINT REFERENCES specializations(id) ON DELETE SET NULL,
    division TEXT,                                -- الشعبة (مثال: شعبة أ، شعبة ب)
    school_name TEXT,                             -- اسم المدرسة
    directorate TEXT,                               -- مديرية التربية والتعليم
    is_trial BOOLEAN DEFAULT FALSE,               -- هل الحساب تجريبي للمعلم (صلاحيات ومخرجات محدودة)
    trial_plans_used INTEGER DEFAULT 0,           -- عدد خطط الدروس المنشأة تجريبياً (الحد الأقصى: 1)
    trial_semester_plans_used INTEGER DEFAULT 0,  -- عدد الخطط الفصلية المنشأة تجريبياً
    subscription_end DATE,
    session_token TEXT,
    session_created_at TIMESTAMP WITH TIME ZONE,
    requested_subjects JSONB,
    approved_by BIGINT REFERENCES users(id),
    approved_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 5. جدول خطط الدروس اليومية
CREATE TABLE IF NOT EXISTS lesson_plans (
    id BIGSERIAL PRIMARY KEY,
    slug TEXT NOT NULL,
    page_start INTEGER NOT NULL,
    page_end INTEGER NOT NULL,
    data JSONB NOT NULL,
    created_by BIGINT REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 6. جدول الخطط الفصلية
CREATE TABLE IF NOT EXISTS semester_plans (
    id BIGSERIAL PRIMARY KEY,
    slug TEXT NOT NULL,
    data JSONB NOT NULL,
    created_by BIGINT REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 7. جدول تحليل المحتوى
CREATE TABLE IF NOT EXISTS content_analysis (
    id BIGSERIAL PRIMARY KEY,
    slug TEXT NOT NULL,
    data JSONB NOT NULL,
    created_by BIGINT REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 8. جدول بنك الأسئلة
CREATE TABLE IF NOT EXISTS questions (
    id BIGSERIAL PRIMARY KEY,
    slug TEXT NOT NULL,
    page_start INTEGER NOT NULL,
    page_end INTEGER NOT NULL,
    qtype TEXT NOT NULL,
    data JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 9. جدول حدود الاستخدام اليومي
CREATE TABLE IF NOT EXISTS daily_usage (
    user_id BIGINT REFERENCES users(id) ON DELETE CASCADE,
    usage_day DATE NOT NULL,
    count INTEGER NOT NULL DEFAULT 0,
    PRIMARY KEY (user_id, usage_day)
);

-- 10. جدول حدود الأدوار
CREATE TABLE IF NOT EXISTS role_limits (
    role TEXT PRIMARY KEY,
    daily_limit INTEGER NOT NULL
);
